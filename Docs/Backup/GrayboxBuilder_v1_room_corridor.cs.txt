using System.Collections.Generic;
using Ib.Core;
using Ib.Environment;
using Ib.Player;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using UnityEngine.SceneManagement;

namespace Ib.EditorTools
{
    /// <summary>
    /// 灰盒关卡生成器。所有尺寸都是常量，重复执行结果一致（覆盖式重建，不做增量）。
    /// 菜单：Tools/Ib/生成灰盒关卡
    /// 命令行：Unity -batchmode -quit -projectPath ... -executeMethod Ib.EditorTools.GrayboxBuilder.BuildFromCommandLine
    /// </summary>
    public static class GrayboxBuilder
    {
        private const string ScenePath = "Assets/Scenes/Graybox.unity";
        private const string MaterialDir = "Assets/Art/Materials";
        private const string PrefabDir = "Assets/Prefabs";
        private const string PlayerPrefabPath = PrefabDir + "/Player.prefab";
        private const string VolumeProfilePath = "Assets/Settings/GrayboxVolume.asset";

        // ---- 尺寸常量（改这里就能改整个关卡）----
        private const float WallThickness = 0.2f;
        private const float RoomHalf = 2.5f;
        private const float RoomHeight = 2.8f;

        private const float CorridorHalfWidth = 0.8f;
        private const float CorridorStartZ = RoomHalf;
        private const float CorridorEndZ = 10.5f;
        private const float CorridorHeight = 2.6f;

        private const float BranchEndX = 4.3f;
        private const float BranchStartZ = 8.9f;

        private const float DoorWidth = 1.0f;
        private const float DoorHeight = 2.1f;

        private static readonly Vector3 SpawnPosition = new Vector3(0f, 0.02f, -1.5f);

        [MenuItem("Tools/Ib/生成灰盒关卡")]
        public static void BuildFromMenu()
        {
            bool ok = EditorUtility.DisplayDialog(
                "生成灰盒关卡",
                "将重建 " + ScenePath + "（覆盖已有内容）并重新生成 Player 预制体。\n\n" +
                "手改过的场景内容会丢失，继续？",
                "生成", "取消");

            if (!ok)
            {
                return;
            }

            Build();
            EditorUtility.DisplayDialog("完成", "灰盒关卡已生成：\n" + ScenePath, "好");
        }

        public static void BuildFromCommandLine()
        {
            Build();
        }

        private static void Build()
        {
            EnsureFolder("Assets/Scenes");
            EnsureFolder("Assets/Art");
            EnsureFolder(MaterialDir);
            EnsureFolder(PrefabDir);

            Material floorRoom = GetOrCreateMaterial("M_Floor_Room", new Color(0.13f, 0.12f, 0.11f), 0.22f);
            Material floorCorridor = GetOrCreateMaterial("M_Floor_Corridor", new Color(0.16f, 0.145f, 0.125f), 0.18f);
            Material wall = GetOrCreateMaterial("M_Wall_Concrete", new Color(0.20f, 0.19f, 0.175f), 0.12f);
            Material ceiling = GetOrCreateMaterial("M_Ceiling", new Color(0.09f, 0.088f, 0.085f), 0.05f);

            Scene scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);

            GameObject levelRoot = new GameObject("Level_Geometry");
            Transform level = levelRoot.transform;

            BuildRoom(level, floorRoom, wall, ceiling);
            BuildCorridor(level, floorCorridor, wall, ceiling);

            foreach (Transform child in level)
            {
                GameObjectUtility.SetStaticEditorFlags(child.gameObject, StaticEditorFlags.BatchingStatic);
            }

            BuildSpawnAndPlayer(scene);
            BuildLighting();
            BuildPostProcessing();
            BuildEnvironmentSettings();

            EditorSceneManager.MarkSceneDirty(scene);
            EditorSceneManager.SaveScene(scene, ScenePath);

            AddToBuildSettings();
            AssetDatabase.SaveAssets();
            AssetDatabase.Refresh();

            Debug.Log("[GrayboxBuilder] 完成：" + ScenePath);
        }

        // ---------------------------------------------------------------- 几何

        private static void BuildRoom(Transform parent, Material floorMat, Material wallMat, Material ceilingMat)
        {
            Transform root = NewGroup(parent, "Room");

            // 墙体错缝相接：东西墙止于北墙内表面，南墙只占内部跨度，
            // 让每对相邻面都是"法线相反"而不是"同向共面"，杜绝 z-fighting。
            float half = RoomHalf;                       // 2.5 内表面
            float outer = RoomHalf + WallThickness;      // 2.7 外表面
            float height = RoomHeight;

            AddSlab(root, "Floor", -outer, outer, -outer, outer, -WallThickness, 0f, floorMat);
            AddSlab(root, "Ceiling", -outer, outer, -outer, outer, height, height + WallThickness, ceilingMat);

            AddSlab(root, "Wall_West", -outer, -half, -outer, half, 0f, height, wallMat);
            AddSlab(root, "Wall_East", half, outer, -outer, half, 0f, height, wallMat);
            AddSlab(root, "Wall_South", -half, half, -outer, -half, 0f, height, wallMat);

            // 北墙带门洞：左右两段 + 门楣
            float doorHalf = DoorWidth * 0.5f;
            AddSlab(root, "Wall_North_L", -half, -doorHalf, half, outer, 0f, height, wallMat);
            AddSlab(root, "Wall_North_R", doorHalf, half, half, outer, 0f, height, wallMat);
            AddSlab(root, "Wall_North_Lintel", -doorHalf, doorHalf, half, outer, DoorHeight, height, wallMat);
        }

        private static void BuildCorridor(Transform parent, Material floorMat, Material wallMat, Material ceilingMat)
        {
            Transform root = NewGroup(parent, "Corridor");

            // 全部按"面坐标"定义，保证相邻块只共边、不共面，避免 z-fighting。
            float outerX = CorridorHalfWidth + WallThickness;   // 1.0
            float outerZ = CorridorEndZ + WallThickness;        // 10.7
            float roomOuterZ = RoomHalf + WallThickness;        // 2.7，与大厅地板外沿对齐

            // ---- A 段：沿 +Z ----
            AddSlab(root, "A_Floor", -outerX, outerX, roomOuterZ, outerZ, -WallThickness, 0f, floorMat);
            AddSlab(root, "A_Ceiling", -outerX, outerX, roomOuterZ, outerZ, CorridorHeight, CorridorHeight + WallThickness, ceilingMat);

            // 西墙贯通到底，东墙止于分支起点，留出右转口
            AddSlab(root, "A_Wall_West", -outerX, -CorridorHalfWidth, roomOuterZ, CorridorEndZ, 0f, CorridorHeight, wallMat);
            AddSlab(root, "A_Wall_East", CorridorHalfWidth, outerX, roomOuterZ, BranchStartZ - WallThickness, 0f, CorridorHeight, wallMat);

            // ---- B 段：从 A 段尽头右转向 +X ----
            AddSlab(root, "B_Floor", 1.0f, BranchEndX + WallThickness, BranchStartZ - WallThickness, outerZ, -WallThickness, 0f, floorMat);
            AddSlab(root, "B_Ceiling", 1.0f, BranchEndX + WallThickness, BranchStartZ - WallThickness, outerZ, CorridorHeight, CorridorHeight + WallThickness, ceilingMat);

            // 转角内侧补墙（接住 A 段东墙的断口）
            AddSlab(root, "B_Wall_South", CorridorHalfWidth, BranchEndX + WallThickness,
                BranchStartZ - WallThickness, BranchStartZ, 0f, CorridorHeight, wallMat);

            // 尽头封墙
            AddSlab(root, "End_Wall_North", -outerX, BranchEndX + WallThickness,
                CorridorEndZ, outerZ, 0f, CorridorHeight, wallMat);

            AddSlab(root, "End_Wall_East", BranchEndX, BranchEndX + WallThickness,
                BranchStartZ, CorridorEndZ, 0f, CorridorHeight, wallMat);
        }

        /// <summary>按最小/最大面坐标生成一块立方体，避免手工算中心点出错。</summary>
        private static GameObject AddSlab(Transform parent, string name,
            float xMin, float xMax, float zMin, float zMax, float yMin, float yMax, Material mat)
        {
            Vector3 center = new Vector3((xMin + xMax) * 0.5f, (yMin + yMax) * 0.5f, (zMin + zMax) * 0.5f);
            Vector3 size = new Vector3(xMax - xMin, yMax - yMin, zMax - zMin);
            return AddBox(parent, name, center, size, mat);
        }

        private static GameObject AddBox(Transform parent, string name, Vector3 center, Vector3 size, Material mat)
        {
            GameObject go = GameObject.CreatePrimitive(PrimitiveType.Cube);
            go.name = name;
            go.transform.SetParent(parent, false);
            go.transform.localPosition = center;
            go.transform.localScale = size;

            MeshRenderer renderer = go.GetComponent<MeshRenderer>();
            renderer.sharedMaterial = mat;
            renderer.shadowCastingMode = ShadowCastingMode.On;
            renderer.receiveShadows = true;

            return go;
        }

        private static Transform NewGroup(Transform parent, string name)
        {
            GameObject go = new GameObject(name);
            go.transform.SetParent(parent, false);
            return go.transform;
        }

        // ---------------------------------------------------------------- 玩家与出生点

        private static void BuildSpawnAndPlayer(Scene scene)
        {
            GameObject spawnGO = new GameObject("PlayerSpawnPoint");
            spawnGO.transform.position = SpawnPosition;
            spawnGO.transform.rotation = Quaternion.identity;
            PlayerSpawnPoint spawn = spawnGO.AddComponent<PlayerSpawnPoint>();

            GameObject prefabAsset = GetOrCreatePlayerPrefab();
            GameObject playerInstance = (GameObject)PrefabUtility.InstantiatePrefab(prefabAsset, scene);
            playerInstance.transform.position = SpawnPosition;
            playerInstance.transform.rotation = Quaternion.identity;

            GameObject manager = new GameObject("--- Game ---");
            PlayerSpawner spawner = manager.AddComponent<PlayerSpawner>();
            SetRef(spawner, "player", playerInstance.GetComponent<FirstPersonController>());
            SetRef(spawner, "spawnPoint", spawn);
        }

        private static GameObject GetOrCreatePlayerPrefab()
        {
            GameObject root = new GameObject("Player");

            CharacterController cc = root.AddComponent<CharacterController>();
            cc.height = 1.8f;
            cc.radius = 0.35f;
            cc.center = new Vector3(0f, 0.9f, 0f);
            cc.slopeLimit = 45f;
            cc.stepOffset = 0.35f;
            cc.skinWidth = 0.02f;
            cc.minMoveDistance = 0f;

            GameObject cameraRoot = new GameObject("CameraRoot");
            cameraRoot.transform.SetParent(root.transform, false);
            cameraRoot.transform.localPosition = new Vector3(0f, 1.62f, 0f);

            GameObject cameraGO = new GameObject("Main Camera");
            cameraGO.tag = "MainCamera";
            cameraGO.transform.SetParent(cameraRoot.transform, false);

            Camera cam = cameraGO.AddComponent<Camera>();
            cam.nearClipPlane = 0.05f;
            cam.farClipPlane = 40f;
            cam.fieldOfView = 68f;
            cam.clearFlags = CameraClearFlags.SolidColor;
            cam.backgroundColor = new Color(0.008f, 0.008f, 0.01f, 1f);
            cam.allowHDR = true;

            cameraGO.AddComponent<AudioListener>();

            UniversalAdditionalCameraData cameraData = cameraGO.AddComponent<UniversalAdditionalCameraData>();
            cameraData.renderPostProcessing = true;

            GameObject hand = new GameObject("HandAnchor");
            hand.transform.SetParent(cameraGO.transform, false);
            hand.transform.localPosition = new Vector3(0.22f, -0.18f, 0.35f);

            FirstPersonController controller = root.AddComponent<FirstPersonController>();
            PlayerLook look = root.AddComponent<PlayerLook>();

            SetRef(controller, "cameraRoot", cameraRoot.transform);
            SetRef(look, "cameraTransform", cameraGO.transform);

            GameObject prefab = PrefabUtility.SaveAsPrefabAsset(root, PlayerPrefabPath);
            Object.DestroyImmediate(root);

            return prefab;
        }

        // ---------------------------------------------------------------- 灯光

        private static void BuildLighting()
        {
            GameObject root = new GameObject("--- Lighting ---");

            AddPointLight(root.transform, "Light_Room", new Vector3(0f, RoomHeight - 0.25f, -0.3f),
                new Color(1f, 0.90f, 0.78f), 2.6f, 8f, false);

            AddPointLight(root.transform, "Light_Corridor_A", new Vector3(0f, CorridorHeight - 0.2f, 6.2f),
                new Color(1f, 0.88f, 0.72f), 2.0f, 7f, true);

            AddPointLight(root.transform, "Light_Corridor_B", new Vector3(2.6f, CorridorHeight - 0.2f, 9.7f),
                new Color(1f, 0.86f, 0.70f), 1.8f, 6f, true);
        }

        private static void AddPointLight(Transform parent, string name, Vector3 position,
            Color color, float intensity, float range, bool flicker)
        {
            GameObject go = new GameObject(name);
            go.transform.SetParent(parent, false);
            go.transform.localPosition = position;

            Light light = go.AddComponent<Light>();
            light.type = LightType.Point;
            light.color = color;
            light.intensity = intensity;
            light.range = range;
            light.shadows = LightShadows.Soft;
            light.shadowStrength = 0.9f;
            light.shadowBias = 0.05f;
            light.shadowNormalBias = 0.4f;

            if (flicker)
            {
                go.AddComponent<LightFlicker>();
            }
        }

        // ---------------------------------------------------------------- 后处理

        private static void BuildPostProcessing()
        {
            VolumeProfile profile = AssetDatabase.LoadAssetAtPath<VolumeProfile>(VolumeProfilePath);

            if (profile == null)
            {
                profile = ScriptableObject.CreateInstance<VolumeProfile>();
                profile.name = "GrayboxVolume";
                AssetDatabase.CreateAsset(profile, VolumeProfilePath);

                Tonemapping tonemapping = AddOverride<Tonemapping>(profile);
                tonemapping.mode.Override(TonemappingMode.ACES);

                ColorAdjustments color = AddOverride<ColorAdjustments>(profile);
                color.postExposure.Override(-0.25f);
                color.contrast.Override(12f);
                color.saturation.Override(-22f);
                color.colorFilter.Override(new Color(0.94f, 0.96f, 1f));

                Vignette vignette = AddOverride<Vignette>(profile);
                vignette.intensity.Override(0.45f);
                vignette.smoothness.Override(0.55f);

                Bloom bloom = AddOverride<Bloom>(profile);
                bloom.threshold.Override(0.95f);
                bloom.intensity.Override(0.55f);
                bloom.scatter.Override(0.6f);

                FilmGrain grain = AddOverride<FilmGrain>(profile);
                grain.type.Override(FilmGrainLookup.Thin1);
                grain.intensity.Override(0.6f);
                grain.response.Override(0.85f);
            }

            GameObject go = new GameObject("Global Volume");
            Volume volume = go.AddComponent<Volume>();
            volume.isGlobal = true;
            volume.priority = 0f;
            volume.profile = profile;
        }

        private static T AddOverride<T>(VolumeProfile profile) where T : VolumeComponent
        {
            T component = ScriptableObject.CreateInstance<T>();
            component.name = typeof(T).Name;
            component.hideFlags = HideFlags.HideInInspector;
            profile.components.Add(component);
            AssetDatabase.AddObjectToAsset(component, profile);
            return component;
        }

        // ---------------------------------------------------------------- 场景环境

        private static void BuildEnvironmentSettings()
        {
            RenderSettings.ambientMode = AmbientMode.Flat;
            RenderSettings.ambientLight = new Color(0.028f, 0.030f, 0.036f);
            RenderSettings.skybox = null;

            RenderSettings.fog = true;
            RenderSettings.fogMode = FogMode.ExponentialSquared;
            RenderSettings.fogDensity = 0.035f;
            RenderSettings.fogColor = new Color(0.02f, 0.02f, 0.025f);

            RenderSettings.reflectionIntensity = 0.3f;
        }

        // ---------------------------------------------------------------- 资产工具

        private static Material GetOrCreateMaterial(string name, Color color, float smoothness)
        {
            string path = MaterialDir + "/" + name + ".mat";
            Material existing = AssetDatabase.LoadAssetAtPath<Material>(path);

            // 已存在就不覆盖：允许之后手动调参而不被重建冲掉
            if (existing != null)
            {
                return existing;
            }

            Shader shader = Shader.Find("Universal Render Pipeline/Lit");
            if (shader == null)
            {
                Debug.LogError("[GrayboxBuilder] 找不到 URP Lit 着色器，检查渲染管线设置。");
                shader = Shader.Find("Standard");
            }

            Material mat = new Material(shader);
            mat.name = name;
            mat.SetColor("_BaseColor", color);
            mat.SetFloat("_Smoothness", smoothness);
            mat.SetFloat("_Metallic", 0f);
            AssetDatabase.CreateAsset(mat, path);
            return mat;
        }

        private static void SetRef(Object target, string fieldName, Object value)
        {
            SerializedObject so = new SerializedObject(target);
            SerializedProperty property = so.FindProperty(fieldName);

            if (property == null)
            {
                Debug.LogError("[GrayboxBuilder] 字段不存在：" + target.GetType().Name + "." + fieldName);
                return;
            }

            property.objectReferenceValue = value;
            so.ApplyModifiedPropertiesWithoutUndo();
            EditorUtility.SetDirty(target);
        }

        private static void EnsureFolder(string path)
        {
            if (AssetDatabase.IsValidFolder(path))
            {
                return;
            }

            string parent = System.IO.Path.GetDirectoryName(path).Replace("\\", "/");
            string leaf = System.IO.Path.GetFileName(path);

            if (!string.IsNullOrEmpty(parent) && !AssetDatabase.IsValidFolder(parent))
            {
                EnsureFolder(parent);
            }

            AssetDatabase.CreateFolder(parent, leaf);
        }

        private static void AddToBuildSettings()
        {
            List<EditorBuildSettingsScene> scenes = new List<EditorBuildSettingsScene>(EditorBuildSettings.scenes);

            for (int i = 0; i < scenes.Count; i++)
            {
                if (scenes[i].path == ScenePath)
                {
                    EditorBuildSettings.scenes = scenes.ToArray();
                    return;
                }
            }

            scenes.Insert(0, new EditorBuildSettingsScene(ScenePath, true));
            EditorBuildSettings.scenes = scenes.ToArray();
        }
    }
